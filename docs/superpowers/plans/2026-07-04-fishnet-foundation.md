> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Two game instances on one PC (or LAN) fly, grapple, and zip around the same arena with server-authoritative movement + client-side prediction. No gameplay added.

**Architecture:** `MovementCore` stays the single source of movement truth. It gains a state snapshot (capture/apply) so Fish-Net can reconcile it. `PlayerMotor` is replaced by `NetworkPlayerMotor : TickNetworkBehaviour` — input collected in `Update` (owner only), simulation runs in `[Replicate]` on Fish-Net's tick, server truth restored in `[Reconcile]`. The player becomes a prefab spawned by Fish-Net's `PlayerSpawner`; a minimal OnGUI `ConnectHUD` does Host/Join over Tugboat (UDP).

**Tech Stack:** Fish-Net 4.7.2 (imported at `Assets/FishNet`, free tier), Tugboat transport, Unity 6 CharacterController.

```csharp
using FishNet.Object.Prediction;
using FishNet.Transporting;
using FishNet.Utility.Template;
using UnityEngine;
using UnityEngine.InputSystem;

namespace AlienSkies.Movement
{
    /// <summary>Networked player motor with client-side prediction. Input is read in
    /// Update (owner only) and accumulated; simulation happens exclusively inside
    /// [Replicate] on the network tick so server and client replay identically;
    /// [Reconcile] restores server truth via MovementCore's state snapshot.</summary>
    [RequireComponent(typeof(CharacterController))]
    public sealed class NetworkPlayerMotor : TickNetworkBehaviour
    {
        public struct OneTimeInput
        {
            public bool JumpPressed;
            public bool GrappleStarted;
            public Vector3 GrappleAnchor;
            public bool GrappleStopped;
            public bool ZipPressed;

            public void ResetState()
            {
                JumpPressed = false;
                GrappleStarted = false;
                GrappleAnchor = default;
                GrappleStopped = false;
                ZipPressed = false;
            }
        }

        public struct ReplicateData : IReplicateData
        {
            public Vector3 WishDir; // world-space, owner-resolved
            public float Yaw;       // for remote visual orientation
            public OneTimeInput OneTime;
            uint _tick;

            public ReplicateData(Vector3 wishDir, float yaw, OneTimeInput oneTime)
            {
                WishDir = wishDir;
                Yaw = yaw;
                OneTime = oneTime;
                _tick = 0;
            }

            public void Dispose() { }
            public uint GetTick() => _tick;
            public void SetTick(uint value) => _tick = value;
        }

        public struct ReconcileData : IReconcileData
        {
            public Vector3 Position;
            public MovementCoreState CoreState;
            uint _tick;

            public ReconcileData(Vector3 position, MovementCoreState coreState)
            {
                Position = position;
                CoreState = coreState;
                _tick = 0;
            }

            public void Dispose() { }
            public uint GetTick() => _tick;
            public void SetTick(uint value) => _tick = value;
        }

        static readonly Vector3 SpawnPoint = new Vector3(0f, 3f, 0f);

        public MovementConfig Config;
        public Transform CameraPivot;
        [SerializeField] float _mouseSensitivity = 0.08f;

        public MovementCore Core { get; private set; }
        public bool HasGrappleTarget { get; private set; }

        CharacterController _cc;
        LineRenderer _rope;
        float _pitch;
        OneTimeInput _oneTime;

        MovementSettings S => Config != null ? Config.Settings : _fallback;
        readonly MovementSettings _fallback = new MovementSettings();

        void Awake()
        {
            _cc = GetComponent<CharacterController>();
            Core = new MovementCore(S);
            BuildRope();
            SetTickCallbacks(TickCallback.Tick | TickCallback.PostTick);
        }

        void BuildRope()
        {
            _rope = gameObject.AddComponent<LineRenderer>();
            _rope.positionCount = 2;
            _rope.startWidth = 0.08f;
            _rope.endWidth = 0.03f;
            // HDR unlit so bloom makes the rope read as an energy line
            var ropeMat = new Material(Shader.Find("Universal Render Pipeline/Unlit"));
            ropeMat.SetColor("_BaseColor", new Color(0.29f, 1f, 0.69f) * 2.6f);
            _rope.material = ropeMat;
            _rope.enabled = false;
        }

        public override void OnStartClient()
        {
            // Only the owner looks through this player's camera or draws its HUD.
            CameraPivot.gameObject.SetActive(IsOwner);
            var hud = GetComponent<MovementHUD>();
            if (hud != null) hud.enabled = IsOwner;
            // Owner shouldn't see the inside of their own body capsule.
            var body = transform.Find("Body");
            if (body != null && IsOwner) body.GetComponent<Renderer>().enabled = false;
            if (IsOwner) Cursor.lockState = CursorLockMode.Locked;
        }

        void Update()
        {
            // Rope renders on every client from replicated simulation state.
            _rope.enabled = Core.IsGrappling;
            if (Core.IsGrappling)
            {
                _rope.SetPosition(0, transform.position + Vector3.down * 0.3f);
                _rope.SetPosition(1, Core.GrappleAnchor);
            }

            if (!IsOwner) return;
            var kb = Keyboard.current;
            var mouse = Mouse.current;
            if (kb == null || mouse == null) return;

            if (kb.escapeKey.wasPressedThisFrame)
                Cursor.lockState = Cursor.lockState == CursorLockMode.Locked
                    ? CursorLockMode.None : CursorLockMode.Locked;

            bool cursorLocked = Cursor.lockState == CursorLockMode.Locked;

            if (cursorLocked)
            {
                Vector2 look = mouse.delta.ReadValue() * _mouseSensitivity;
                transform.Rotate(0f, look.x, 0f);
                _pitch = Mathf.Clamp(_pitch - look.y, -89f, 89f);
                CameraPivot.localRotation = Quaternion.Euler(_pitch, 0f, 0f);
            }

            // Accumulate one-time inputs between ticks (consumed in BuildData).
            Vector3 anchor = default;
            HasGrappleTarget = cursorLocked && GrappleAimer.TryFindAnchor(
                CameraPivot.position, CameraPivot.forward, S, transform, out anchor);
            if (cursorLocked && kb.fKey.wasPressedThisFrame && HasGrappleTarget)
            {
                _oneTime.GrappleStarted = true;
                _oneTime.GrappleAnchor = anchor;
            }
            if (!cursorLocked || kb.fKey.wasReleasedThisFrame)
                _oneTime.GrappleStopped = true;
            if (cursorLocked && mouse.rightButton.wasPressedThisFrame)
                _oneTime.ZipPressed = true;
            if (cursorLocked && kb.spaceKey.wasPressedThisFrame)
                _oneTime.JumpPressed = true;
        }

        protected override void TimeManager_OnTick() => PerformReplicate(BuildData());

        protected override void TimeManager_OnPostTick() => CreateReconcile();

        ReplicateData BuildData()
        {
            if (!IsOwner) return default;

            var kb = Keyboard.current;
            Vector3 wish = Vector3.zero;
            if (kb != null && Cursor.lockState == CursorLockMode.Locked)
            {
                float f = (kb.wKey.isPressed ? 1f : 0f) - (kb.sKey.isPressed ? 1f : 0f);
                float r = (kb.dKey.isPressed ? 1f : 0f) - (kb.aKey.isPressed ? 1f : 0f);
                wish = transform.forward * f + transform.right * r;
                wish.y = 0f;
                if (wish.sqrMagnitude > 1e-4f) wish.Normalize();
            }

            var data = new ReplicateData(wish, transform.eulerAngles.y, _oneTime);
            _oneTime.ResetState();
            return data;
        }

        [Replicate]
        void PerformReplicate(ReplicateData rd, ReplicateState state = ReplicateState.Invalid, Channel channel = Channel.Unreliable)
        {
            float delta = (float)TimeManager.TickDelta;

            if (!IsOwner) // owner already rotated in Update; remotes follow replicated yaw
                transform.rotation = Quaternion.Euler(0f, rd.Yaw, 0f);

            if (rd.OneTime.GrappleStarted) Core.StartGrapple(rd.OneTime.GrappleAnchor);
            if (rd.OneTime.GrappleStopped) Core.StopGrapple();
            if (rd.OneTime.ZipPressed) Core.StartZip();

            var input = new MoveInput { WishDir = rd.WishDir, JumpPressed = rd.OneTime.JumpPressed };
            Core.Tick(input, transform.position, _cc.isGrounded, delta);
            _cc.Move(Core.Velocity * delta);

            if (transform.position.y < S.KillY)
                Teleport(SpawnPoint); // deterministic: replays reach the same y the same tick
        }

        [Reconcile]
        void PerformReconcile(ReconcileData rd, Channel channel = Channel.Unreliable)
        {
            Core.ApplyState(rd.CoreState);
            Teleport(rd.Position);
        }

        public override void CreateReconcile() =>
            PerformReconcile(new ReconcileData(transform.position, Core.CaptureState()));

        void Teleport(Vector3 position)
        {
            // CharacterController must be disabled around direct position writes or
            // its internal physics state stays at the old position until next simulate.
            _cc.enabled = false;
            transform.position = position;
            _cc.enabled = true;
            if (position == SpawnPoint) Core.Reset();
        }

        void OnControllerColliderHit(ControllerColliderHit hit) =>
            Core.ClampAgainstSurface(hit.normal);
    }
}
```
