```csharp
namespace AlienSkies.Combat
{
    /// <summary>Pure round state machine: first score to reach the frag target
    /// ends the round exactly once; after the reset delay the round restarts.
    /// MatchManager drives it on the server only.</summary>
    public sealed class MatchCore
    {
        readonly CombatSettings _s;

        public bool IsOver { get; private set; }
        public float ResetRemaining { get; private set; }

        public MatchCore(CombatSettings settings) => _s = settings;

        /// <summary>Returns true exactly when this report ends the round.</summary>
        public bool ReportScore(int frags)
        {
            if (IsOver || frags < _s.FragTarget) return false;
            IsOver = true;
            ResetRemaining = _s.RoundResetDelay;
            return true;
        }

        /// <summary>Returns true when the reset delay has elapsed.</summary>
        public bool TickReset(float dt)
        {
            if (!IsOver) return false;
            ResetRemaining -= dt;
            return ResetRemaining <= 0f;
        }

        public void StartNewRound()
        {
            IsOver = false;
            ResetRemaining = 0f;
        }
    }
}
```

```csharp
using FishNet.Object;
using FishNet.Object.Synchronizing;
using UnityEngine;

namespace AlienSkies.Combat
{
    /// <summary>Scene NetworkObject singleton: server watches frag counts, ends
    /// the round at the target, freezes play via IsRoundOver (players poll it),
    /// and resets every PlayerCombat after the delay. Clients read the SyncVars
    /// for the VICTORY/DEFEATED splash.</summary>
    public sealed class MatchManager : NetworkBehaviour
    {
        public static MatchManager Instance { get; private set; }

        public CombatConfig Config;

        readonly SyncVar<bool> _roundOver = new SyncVar<bool>();
        readonly SyncVar<int> _winnerClientId = new SyncVar<int>(-1);

        public bool IsRoundOver => _roundOver.Value;
        public int WinnerClientId => _winnerClientId.Value;
        public float ResetRemaining { get; private set; } // client-side HUD estimate

        MatchCore _core;

        CombatSettings S => Config != null ? Config.Settings : _fallback;
        readonly CombatSettings _fallback = new CombatSettings();

        void Awake()
        {
            Instance = this;
            _core = new MatchCore(S);
            _roundOver.OnChange += HandleRoundOverChanged;
        }

        void OnDestroy()
        {
            _roundOver.OnChange -= HandleRoundOverChanged;
            if (Instance == this) Instance = null;
        }

        void HandleRoundOverChanged(bool prev, bool next, bool asServer)
        {
            if (next && !asServer) ResetRemaining = S.RoundResetDelay;
        }

        void Update()
        {
            float dt = Time.deltaTime;
            if (ResetRemaining > 0f) ResetRemaining -= dt;
            if (!IsServerStarted) return;

            if (!_core.IsOver)
            {
                foreach (var pc in FindObjectsByType<PlayerCombat>(FindObjectsSortMode.None))
                {
                    if (!_core.ReportScore(pc.Frags)) continue;
                    _winnerClientId.Value = pc.OwnerId;
                    _roundOver.Value = true;
                    break;
                }
            }
            else if (_core.TickReset(dt))
            {
                foreach (var pc in FindObjectsByType<PlayerCombat>(FindObjectsSortMode.None))
                    pc.ServerResetForNewRound();
                _core.StartNewRound();
                _roundOver.Value = false;
                _winnerClientId.Value = -1;
            }
        }
    }
}
```
