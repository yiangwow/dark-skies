**Goal:** A first-person movement toy — momentum-preserving Quake-lineage movement plus a zip-grapple — in a graybox cosmic arena, fully unit-tested at the math layer.

```bash
# Run Unity EditMode tests headless from WSL.
# REQUIREMENT: the Unity Editor must NOT have this project open.
set -uo pipefail

UNITY="/mnt/c/Program Files/Unity/Hub/Editor/6000.5.1f1/Editor/Unity.exe"
PROJ_WIN='C:\Users\computer\My project'
PROJ_WSL="/mnt/c/Users/computer/My project"
RESULTS_WIN="${PROJ_WIN}\\TestResults\\editmode.xml"
LOG_WIN="${PROJ_WIN}\\TestResults\\editmode.log"
RESULTS_WSL="${PROJ_WSL}/TestResults/editmode.xml"
LOG_WSL="${PROJ_WSL}/TestResults/editmode.log"

mkdir -p "${PROJ_WSL}/TestResults"
rm -f "${RESULTS_WSL}"

"${UNITY}" -batchmode -projectPath "${PROJ_WIN}" -runTests \
  -testPlatform EditMode -testResults "${RESULTS_WIN}" -logFile "${LOG_WIN}"
UNITY_EXIT=$?

if [[ ! -f "${RESULTS_WSL}" ]]; then
  echo "FAIL: no results file produced (unity exit ${UNITY_EXIT})."
  echo "Likely causes: editor has the project open, or compile error. Log tail:"
  tail -40 "${LOG_WSL}" 2>/dev/null
  grep -m5 "error CS" "${LOG_WSL}" 2>/dev/null
  exit 1
fi

python3 - "${RESULTS_WSL}" <<'EOF'
import sys, xml.etree.ElementTree as ET
root = ET.parse(sys.argv[1]).getroot()
print(f"result={root.get('result')} total={root.get('total')} "
      f"passed={root.get('passed')} failed={root.get('failed')}")
for tc in root.iter('test-case'):
    if tc.get('result') != 'Passed':
        print(f"  FAILED: {tc.get('fullname')}")
        msg = tc.find('.//message')
        if msg is not None and msg.text:
            print("    " + msg.text.strip().replace("\n", "\n    ")[:500])
sys.exit(0 if root.get('result') == 'Passed' else 1)
EOF
```

Create `Assets/AlienSkies/Scripts/MovementCore.cs`:

```csharp
using UnityEngine;

namespace AlienSkies.Movement
{
    /// <summary>Pure movement math. No MonoBehaviour, no Physics — fully unit-testable.
    /// Quake lineage: friction on ground only; acceleration capped along the wish
    /// direction so momentum from grapples/falls is never destroyed by steering.</summary>
    public sealed class MovementCore
    {
        const float GroundStickSpeed = -2f;

        readonly MovementSettings _s;

        public Vector3 Velocity { get; private set; }
        public bool IsGrappling { get; private set; }
        public Vector3 GrappleAnchor { get; private set; }

        public MovementCore(MovementSettings settings) => _s = settings;

        public void SetVelocity(Vector3 v) => Velocity = v;

        public void Reset()
        {
            Velocity = Vector3.zero;
            IsGrappling = false;
        }

        public void Tick(in MoveInput input, Vector3 position, bool grounded, float dt)
        {
            if (grounded && !IsGrappling)
            {
                ApplyGroundFriction(dt);
                Accelerate(input.WishDir, _s.GroundMaxSpeed, _s.GroundAccel, dt);
                if (Velocity.y < GroundStickSpeed)
                    Velocity = new Vector3(Velocity.x, GroundStickSpeed, Velocity.z);
            }
            else
            {
                float accel = IsGrappling ? _s.GrappleSteerAccel : _s.AirAccel;
                Accelerate(input.WishDir, _s.AirControlMaxSpeed, accel, dt);
            }

            if (IsGrappling)
                PullTowardAnchor(position, dt);
            else if (!grounded)
                Velocity += Vector3.down * (_s.Gravity * dt);

            if (grounded && input.JumpPressed)
                Velocity = new Vector3(Velocity.x, _s.JumpSpeed, Velocity.z);

            ApplySoftCap(dt);
        }

        void ApplyGroundFriction(float dt)
        {
            var horiz = new Vector3(Velocity.x, 0f, Velocity.z);
            float speed = horiz.magnitude;
            if (speed <= 0f) return;
            float newSpeed = Mathf.Max(speed - speed * _s.GroundFriction * dt, 0f);
            horiz *= newSpeed / speed;
            Velocity = new Vector3(horiz.x, Velocity.y, horiz.z);
        }

        void Accelerate(Vector3 wishDir, float maxSpeed, float accel, float dt)
        {
            if (wishDir.sqrMagnitude < 1e-6f) return;
            float speedAlong = Vector3.Dot(Velocity, wishDir);
            float addSpeed = maxSpeed - speedAlong;
            if (addSpeed <= 0f) return;
            Velocity += wishDir * Mathf.Min(accel * dt, addSpeed);
        }

        void PullTowardAnchor(Vector3 position, float dt)
        {
            var toAnchor = GrappleAnchor - position;
            if (toAnchor.sqrMagnitude < 1f) return; // arrived — stop pulling, keep momentum
            Velocity += toAnchor.normalized * (_s.GrapplePullAccel * dt);
        }

        void ApplySoftCap(float dt)
        {
            float speed = Velocity.magnitude;
            if (speed <= _s.SoftCapSpeed) return;
            float over = (speed - _s.SoftCapSpeed) / _s.SoftCapSpeed;
            Velocity *= Mathf.Max(1f - _s.SoftCapDrag * over * dt, 0f);
        }

        // --- Grapple API (exercised by Task 5's tests) ---

        public void StartGrapple(Vector3 anchor)
        {
            IsGrappling = true;
            GrappleAnchor = anchor;
        }

        public void StopGrapple()
        {
            if (!IsGrappling) return;
            IsGrappling = false;
            var dir = Velocity.sqrMagnitude > 1e-4f ? Velocity.normalized : Vector3.up;
            Velocity += dir * _s.GrappleReleaseBoost + Vector3.up * _s.GrappleReleaseUpBoost;
        }

        /// <summary>Collision response: remove the velocity component pointing into a
        /// surface (called by the motor from OnControllerColliderHit).</summary>
        public void ClampAgainstSurface(Vector3 normal)
        {
            float into = Vector3.Dot(Velocity, normal);
            if (into < 0f) Velocity -= normal * into;
        }
    }
}
```

```csharp
using UnityEngine;

namespace AlienSkies.Movement
{
    /// <summary>Finds a grapple anchor along the look ray: exact raycast first,
    /// then a sphere cast as aim assist. Skips the player's own colliders.</summary>
    public static class GrappleAimer
    {
        static readonly RaycastHit[] Hits = new RaycastHit[16];

        public static bool TryFindAnchor(Vector3 origin, Vector3 dir, MovementSettings s,
                                         Transform self, out Vector3 anchor)
        {
            if (TryCast(new Ray(origin, dir), 0f, s.GrappleRange, self, out anchor)) return true;
            return TryCast(new Ray(origin, dir), s.AimAssistRadius, s.GrappleRange, self, out anchor);
        }

        static bool TryCast(Ray ray, float radius, float range, Transform self, out Vector3 anchor)
        {
            anchor = default;
            int n = radius <= 0f
                ? Physics.RaycastNonAlloc(ray, Hits, range)
                : Physics.SphereCastNonAlloc(ray, radius, Hits, range);
            float bestDist = float.MaxValue;
            bool found = false;
            for (int i = 0; i < n; i++)
            {
                var h = Hits[i];
                if (h.transform == self || h.transform.IsChildOf(self)) continue;
                if (h.distance < 1f) continue; // spherecast starting overlaps report distance 0
                if (h.distance < bestDist) { bestDist = h.distance; anchor = h.point; found = true; }
            }
            return found;
        }
    }
}
```

```csharp
using UnityEngine;
using UnityEngine.InputSystem;

namespace AlienSkies.Movement
{
    /// <summary>Thin adapter: reads devices, feeds MovementCore, applies velocity via
    /// CharacterController.Move. All feel-critical math lives in MovementCore.</summary>
    [RequireComponent(typeof(CharacterController))]
    public sealed class PlayerMotor : MonoBehaviour
    {
        public MovementConfig Config;
        public Transform CameraPivot;
        [SerializeField] float _mouseSensitivity = 0.08f;

        public MovementCore Core { get; private set; }
        public bool HasGrappleTarget { get; private set; }

        CharacterController _cc;
        LineRenderer _rope;
        float _pitch;
        Vector3 _spawnPos;

        MovementSettings S => Config != null ? Config.Settings : _fallback;
        readonly MovementSettings _fallback = new MovementSettings();

        void Awake()
        {
            _cc = GetComponent<CharacterController>();
            Core = new MovementCore(S);
            _spawnPos = transform.position;
            Cursor.lockState = CursorLockMode.Locked;

            _rope = gameObject.AddComponent<LineRenderer>();
            _rope.positionCount = 2;
            _rope.startWidth = 0.06f;
            _rope.endWidth = 0.02f;
            _rope.material = new Material(Shader.Find("Sprites/Default")) { color = new Color(0.4f, 1f, 0.75f) };
            _rope.enabled = false;
        }

        void Update()
        {
            var kb = Keyboard.current;
            var mouse = Mouse.current;
            if (kb == null || mouse == null) return;

            if (kb.escapeKey.wasPressedThisFrame)
                Cursor.lockState = Cursor.lockState == CursorLockMode.Locked
                    ? CursorLockMode.None : CursorLockMode.Locked;

            // Look
            Vector2 look = mouse.delta.ReadValue() * _mouseSensitivity;
            transform.Rotate(0f, look.x, 0f);
            _pitch = Mathf.Clamp(_pitch - look.y, -89f, 89f);
            CameraPivot.localRotation = Quaternion.Euler(_pitch, 0f, 0f);

            // Wish direction (horizontal, camera-relative yaw)
            float f = (kb.wKey.isPressed ? 1f : 0f) - (kb.sKey.isPressed ? 1f : 0f);
            float r = (kb.dKey.isPressed ? 1f : 0f) - (kb.aKey.isPressed ? 1f : 0f);
            var wish = transform.forward * f + transform.right * r;
            wish.y = 0f;
            if (wish.sqrMagnitude > 1e-4f) wish.Normalize();

            // Grapple target preview (for crosshair) + input
            HasGrappleTarget = GrappleAimer.TryFindAnchor(
                CameraPivot.position, CameraPivot.forward, S, transform, out var anchor);
            if (mouse.rightButton.wasPressedThisFrame && HasGrappleTarget)
                Core.StartGrapple(anchor);
            if (mouse.rightButton.wasReleasedThisFrame)
                Core.StopGrapple();

            var input = new MoveInput { WishDir = wish, JumpPressed = kb.spaceKey.wasPressedThisFrame };
            Core.Tick(input, transform.position, _cc.isGrounded, Time.deltaTime);
            _cc.Move(Core.Velocity * Time.deltaTime);

            // Rope visual
            _rope.enabled = Core.IsGrappling;
            if (Core.IsGrappling)
            {
                _rope.SetPosition(0, transform.position + Vector3.down * 0.3f);
                _rope.SetPosition(1, Core.GrappleAnchor);
            }

            if (transform.position.y < S.KillY) Respawn();
        }

        void OnControllerColliderHit(ControllerColliderHit hit) =>
            Core.ClampAgainstSurface(hit.normal);

        public void Respawn()
        {
            _cc.enabled = false;
            transform.position = _spawnPos;
            _cc.enabled = true;
            Core.Reset();
        }
    }
}
```

```csharp
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using AlienSkies.Movement;

namespace AlienSkies.EditorTools
{
    /// <summary>Builds the graybox MovementPlayground scene from code so it is
    /// rebuildable headlessly (-executeMethod) and reviewable in diffs.</summary>
    public static class PlaygroundBuilder
    {
        const string ScenePath = "Assets/AlienSkies/Scenes/MovementPlayground.unity";
        const string ConfigPath = "Assets/AlienSkies/Config/MovementConfig.asset";

        [MenuItem("AlienSkies/Build Movement Playground")]
        public static void Build()
        {
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);

            var config = AssetDatabase.LoadAssetAtPath<MovementConfig>(ConfigPath);
            if (config == null)
            {
                config = ScriptableObject.CreateInstance<MovementConfig>();
                AssetDatabase.CreateAsset(config, ConfigPath);
            }

            // Dim sun so graybox shapes still read against the dark sky
            var sun = new GameObject("Sun").AddComponent<Light>();
            sun.type = LightType.Directional;
            sun.intensity = 0.7f;
            sun.transform.rotation = Quaternion.Euler(45f, 30f, 0f);
            RenderSettings.ambientMode = UnityEngine.Rendering.AmbientMode.Flat;
            RenderSettings.ambientLight = new Color(0.12f, 0.10f, 0.20f);

            BuildPlayer(config);
            BuildArena();

            EditorSceneManager.SaveScene(scene, ScenePath);
            AssetDatabase.SaveAssets();
            Debug.Log($"PlaygroundBuilder: saved {ScenePath}");
        }

        static void BuildPlayer(MovementConfig config)
        {
            var player = new GameObject("Player");
            player.transform.position = new Vector3(0f, 3f, 0f);
            var cc = player.AddComponent<CharacterController>();
            cc.height = 1.8f;
            cc.radius = 0.4f;
            cc.center = new Vector3(0f, 0.9f, 0f);

            var pivot = new GameObject("CameraPivot").transform;
            pivot.SetParent(player.transform, false);
            pivot.localPosition = new Vector3(0f, 1.6f, 0f);

            var camGO = new GameObject("MainCamera") { tag = "MainCamera" };
            camGO.transform.SetParent(pivot, false);
            var cam = camGO.AddComponent<Camera>();
            cam.clearFlags = CameraClearFlags.SolidColor;
            cam.backgroundColor = new Color(0.015f, 0.008f, 0.045f); // void violet
            cam.farClipPlane = 2000f;
            camGO.AddComponent<AudioListener>();

            var motor = player.AddComponent<PlayerMotor>();
            motor.Config = config;
            motor.CameraPivot = pivot;
            var hud = player.AddComponent<MovementHUD>();
            hud.Motor = motor;
        }

        static void BuildArena()
        {
            // Spawn platform
            CreateBox("SpawnPlatform", new Vector3(0f, 0f, 0f), new Vector3(24f, 1f, 24f));

            // Galaxy stand-ins: deterministic field of spheres in a shell around spawn
            var rng = new System.Random(42);
            for (int i = 0; i < 42; i++)
            {
                float yaw = (float)(rng.NextDouble() * Mathf.PI * 2.0);
                float dist = 35f + (float)rng.NextDouble() * 220f;
                float height = -10f + (float)rng.NextDouble() * 120f;
                var pos = new Vector3(Mathf.Cos(yaw) * dist, height, Mathf.Sin(yaw) * dist);

                var sphere = GameObject.CreatePrimitive(PrimitiveType.Sphere);
                sphere.name = $"Galaxy_{i:00}";
                sphere.transform.position = pos;
                sphere.transform.localScale = Vector3.one * (5f + (float)rng.NextDouble() * 28f);
            }

            // A few flat rest platforms at increasing heights
            for (int i = 1; i <= 6; i++)
            {
                float yaw = i * 1.9f;
                var pos = new Vector3(Mathf.Cos(yaw), 0f, Mathf.Sin(yaw)) * (40f + i * 25f)
                          + Vector3.up * (i * 14f);
                CreateBox($"Platform_{i}", pos, new Vector3(10f, 0.8f, 10f));
            }
        }

        static void CreateBox(string name, Vector3 pos, Vector3 scale)
        {
            var box = GameObject.CreatePrimitive(PrimitiveType.Cube);
            box.name = name;
            box.transform.position = pos;
            box.transform.localScale = scale;
        }
```
